func TestAuditRuntimepathDispatch(t *testing.T) {
	s := New(nil, nil, io.Discard)
	t.Cleanup(s.stopAnalysis)
	root := t.TempDir()
	registered, publishHeld, reached := -1, false, false
	s.testHooks.discoverWorkspaceFiles = func(context.Context, string, int) ([]string, bool, error) {
		reached = true
		s.mu.Lock()
		registered = len(s.cancellations)
		s.mu.Unlock()
		publishHeld = !s.publishMu.TryLock()
		if !publishHeld {
			s.publishMu.Unlock()
		}
		return nil, false, nil
	}
	body, err := json.Marshal(map[string]any{"jsonrpc": "2.0", "id": 42, "method": MethodDidChangeRuntimepath, "params": map[string]any{"runtimepath": []string{root}}})
	if err != nil {
		t.Fatal(err)
	}
	input := encodeFrames(t, `{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"initializationOptions":{"runtimepath":[]}}}`, string(body), `{"jsonrpc":"2.0","id":2,"method":"shutdown"}`, `{"jsonrpc":"2.0","method":"exit"}`)
	var output bytes.Buffer
	s.input, s.output = &input, &output
	if code := s.Run(context.Background()); code != 0 {
		t.Fatalf("exit code=%d", code)
	}
	if !reached || registered != 0 || !publishHeld {
		t.Fatalf("unexpected observation: reached=%v registered=%d publishHeld=%v", reached, registered, publishHeld)
	}
	t.Logf("runtimepath request id=42: registered=%d publishMuHeldDuringDiscovery=%v", registered, publishHeld)
}

func TestAuditMultiRootResolver(t *testing.T) {
	first, second, runtimeRoot := t.TempDir(), t.TempDir(), t.TempDir()
	lib := writeWorkspaceFile(t, second, "lib.vim", "vim9script\nexport var value = 1\n")
	from := filepath.Join(second, "main.vim")
	resolver := workspacePathResolver([]string{first, second}, []string{runtimeRoot})
	got := resolver.ResolveImportPath(from, "'./lib.vim'", false)
	allRoots, err := workspace.NewPathResolverForRoots([]string{first, second}, []string{runtimeRoot})
	if err != nil {
		t.Fatal(err)
	}
	want := allRoots.ResolveImportPath(from, "'./lib.vim'", false)
	if got.Path != "" || want.Path == "" || !sameWorkspacePath(want.Path, mustWorkspaceCanonicalPath(t, lib)) {
		t.Fatalf("server=%+v allRoots=%+v", got, want)
	}
	t.Logf("server resolver: path=%q candidates=%d; existing all-roots API resolves lib.vim", got.Path, len(got.Candidates))
}

func TestAuditRenameConflict(t *testing.T) {
	source := "vim9script\nvar first = 1\nvar other = 2\necho first + other\n"
	s, documentURI := openNavigationDocument(t, text.UTF16, source)
	t.Cleanup(s.stopAnalysis)
	edit, err := s.Rename(context.Background(), &protocol.RenameParams{
		TextDocumentPositionParams: protocol.TextDocumentPositionParams{TextDocument: protocol.TextDocumentIdentifier{URI: documentURI}, Position: protocol.Position{Line: 1, Character: 5}},
		NewName:                    "other",
	})
	if err != nil || edit == nil {
		t.Fatalf("rename=%#v err=%v", edit, err)
	}
	before := text.NewSnapshot(documentURI.String(), 1, nil, source)
	after := source
	edits := edit.DocumentChanges[0].(*protocol.TextDocumentEdit).Edits
	for i := len(edits) - 1; i >= 0; i-- {
		e := edits[i].(*protocol.TextEdit)
		start, _ := before.Offset(fromProtocolPosition(e.Range.Start), text.UTF16)
		end, _ := before.Offset(fromProtocolPosition(e.Range.End), text.UTF16)
		after = after[:start] + e.NewText + after[end:]
	}
	file := syntax.Parse(after)
	codes := make([]string, 0)
	for _, d := range analysis.CombinedDiagnostics(file, analysis.Analyze(file)) {
		codes = append(codes, d.Code)
	}
	if !strings.Contains(after, "var other = 1\nvar other = 2") || len(codes) == 0 {
		t.Fatalf("after=%q codes=%v", after, codes)
	}
	t.Logf("rename accepted conflicting name; after=%q diagnosticCodes=%v", after, codes)
}

func TestAuditRenameClosedDiskStale(t *testing.T) {
	root := t.TempDir()
	old := "vim9script\nexport def Run()\nenddef\n"
	lib := writeWorkspaceFile(t, root, "lib.vim", old)
	source := "vim9script\nimport './lib.vim' as lib\nlib.Run()\n"
	main := writeWorkspaceFile(t, root, "main.vim", source)
	s := initializeWorkspaceServer(t, root)
	documentURI := uri.File(main)
	s.documents.Open(documentURI.String(), 1, source)
	changed := "# external edit\n" + old
	if err := os.WriteFile(lib, []byte(changed), 0o600); err != nil {
		t.Fatal(err)
	}
	edit, err := s.Rename(context.Background(), &protocol.RenameParams{
		TextDocumentPositionParams: protocol.TextDocumentPositionParams{TextDocument: protocol.TextDocumentIdentifier{URI: documentURI}, Position: protocol.Position{Line: 2, Character: 5}}, NewName: "Execute",
	})
	if err != nil || edit == nil {
		t.Fatalf("rename=%#v err=%v", edit, err)
	}
	for _, change := range edit.DocumentChanges {
		document := change.(*protocol.TextDocumentEdit)
		if !sameWorkspacePath(document.TextDocument.URI.FsPath(), mustWorkspaceCanonicalPath(t, lib)) {
			continue
		}
		e := document.Edits[0].(*protocol.TextEdit)
		snapshot := text.NewSnapshot(document.TextDocument.URI.String(), 0, nil, changed)
		start, startErr := snapshot.Offset(fromProtocolPosition(e.Range.Start), text.UTF16)
		end, endErr := snapshot.Offset(fromProtocolPosition(e.Range.End), text.UTF16)
		t.Logf("closed-file rename accepted stale range=%+v version=%v currentOffsetErrors=(%v,%v)", e.Range, document.TextDocument.Version, startErr, endErr)
		if startErr == nil && endErr == nil && snapshot.Text()[start:end] == "Run" {
			t.Fatal("range unexpectedly current")
		}
		return
	}
	t.Fatal("missing closed-file edit")
}

type auditConfigurationClient struct {
	protocol.UnimplementedClient
	started chan struct{}
	release chan struct{}
}

func (c *auditConfigurationClient) Configuration(context.Context, *protocol.ConfigurationParams) ([]protocol.LSPAny, error) {
	close(c.started)
	<-c.release
	return []protocol.LSPAny{protocol.LSPAny(`{"diagnostic":{"maxNumber":5}}`)}, nil
}

func TestAuditConfigurationOutOfOrder(t *testing.T) {
	s := New(nil, nil, io.Discard)
	t.Cleanup(s.stopAnalysis)
	client := &auditConfigurationClient{started: make(chan struct{}), release: make(chan struct{})}
	s.client, s.workspaceConfiguration = client, true
	done := make(chan error, 1)
	go func() { done <- s.refreshWorkspaceConfiguration(context.Background()) }()
	<-client.started
	if err := s.applyWorkspaceConfiguration(context.Background(), []byte(`{"diagnostic":{"maxNumber":99}}`)); err != nil {
		t.Fatal(err)
	}
	close(client.release)
	if err := <-done; err != nil {
		t.Fatal(err)
	}
	if s.diagnosticMaxNumber != 5 {
		t.Fatalf("maxNumber=%d", s.diagnosticMaxNumber)
	}
	t.Log("new maxNumber=99 overwritten by old workspace/configuration response maxNumber=5")
}

func TestAuditWatchedUpdateAfterRebuild(t *testing.T) {
	root := t.TempDir()
	path := writeWorkspaceFile(t, root, "lib.vim", "function Initial()\nendfunction\n")
	s := initializeWorkspaceServer(t, root)
	oldEvent := "function OldEvent()\nendfunction\n"
	latest := "function Latest()\nendfunction\n"
	if err := os.WriteFile(path, []byte(oldEvent), 0o600); err != nil {
		t.Fatal(err)
	}
	paused, release := make(chan struct{}), make(chan struct{})
	s.testHooks.beforeWatchedFileInstall = func(string) { close(paused); <-release }
	done := make(chan bool, 1)
	go func() {
		done <- s.applyWatchedFileChanges(context.Background(), []protocol.FileEvent{{URI: uri.File(path), Type: protocol.FileChangeTypeChanged}})
	}()
	<-paused
	if err := os.WriteFile(path, []byte(latest), 0o600); err != nil {
		t.Fatal(err)
	}
	s.scheduleWorkspaceRebuild()
	s.workspaceWG.Wait()
	canonical := mustWorkspaceCanonicalPath(t, path)
	before, _ := s.workspaceIndex.Source(canonical)
	close(release)
	if !<-done {
		t.Fatal("unexpected fallback")
	}
	after, _ := s.workspaceIndex.Source(canonical)
	if before != latest || after != oldEvent {
		t.Fatalf("before=%q after=%q", before, after)
	}
	t.Log("new rebuild installed Latest(); older watched-file read then overwrote it with OldEvent()")
}

func TestAuditCloseFIFO(t *testing.T) {
	root := t.TempDir()
	path := writeWorkspaceFile(t, root, "closed.vim", "let value = 1\n")
	s := initializeWorkspaceServer(t, root)
	documentURI := uri.File(path)
	s.documents.Open(documentURI.String(), 1, "let value = 1\n")
	if err := os.Remove(path); err != nil {
		t.Fatal(err)
	}
	if err := syscall.Mkfifo(path, 0o600); err != nil {
		t.Fatal(err)
	}
	entered := make(chan struct{})
	s.testHooks.beforeWorkspaceRestoreRead = func(workspaceRestore) { close(entered) }
	done := make(chan error, 1)
	go func() {
		done <- s.DidClose(context.Background(), &protocol.DidCloseTextDocumentParams{TextDocument: protocol.TextDocumentIdentifier{URI: documentURI}})
	}()
	<-entered
	s.cancelAnalysis()
	select {
	case err := <-done:
		t.Fatalf("close unexpectedly returned before opening FIFO writer: %v", err)
	case <-time.After(100 * time.Millisecond):
	}
	writer, err := os.OpenFile(path, os.O_RDWR|syscall.O_NONBLOCK, 0)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := writer.WriteString("let value = 1\n"); err != nil {
		t.Fatal(err)
	}
	if err := writer.Close(); err != nil {
		t.Fatal(err)
	}
	select {
	case err := <-done:
		if err != nil {
			t.Fatal(err)
		}
	case <-time.After(5 * time.Second):
		t.Fatal("close did not finish after FIFO release")
	}
	t.Log("didClose stayed blocked on FIFO after lifecycle cancellation; completed only after a writer supplied data and closed")
}

func TestAuditClientCapabilityFallback(t *testing.T) {
	s := New(nil, nil, io.Discard)
	t.Cleanup(s.stopAnalysis)
	if _, err := s.Initialize(context.Background(), &protocol.InitializeParams{InitializationOptions: protocol.LSPAny(`{"runtimepath":[]}`)}); err != nil {
		t.Fatal(err)
	}
	documentURI := uri.File(filepath.Join(t.TempDir(), "main.vim"))
	s.documents.Open(documentURI.String(), 1, "vim9script\nvar value = 1\necho value\n")
	edit, err := s.Rename(context.Background(), &protocol.RenameParams{TextDocumentPositionParams: protocol.TextDocumentPositionParams{
		TextDocument: protocol.TextDocumentIdentifier{URI: documentURI}, Position: protocol.Position{Line: 1, Character: 5},
	}, NewName: "renamed"})
	if err != nil || edit == nil || len(edit.DocumentChanges) == 0 || len(edit.Changes) != 0 {
		t.Fatalf("edit=%#v err=%v", edit, err)
	}
	// Use a function to guarantee a non-empty document-symbol response.
	s.documents.Open(documentURI.String(), 2, "vim9script\ndef Run()\nenddef\n")
	symbols, err := s.DocumentSymbol(context.Background(), &protocol.DocumentSymbolParams{TextDocument: protocol.TextDocumentIdentifier{URI: documentURI}})
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := symbols.(protocol.DocumentSymbolSlice); !ok {
		t.Fatalf("symbols=%T", symbols)
	}
	t.Log("client advertised neither documentChanges nor hierarchicalDocumentSymbolSupport; server returned versioned DocumentChanges and DocumentSymbolSlice")
}

func TestAuditPinnedVimOracle(t *testing.T) {
	vim := os.Getenv("VIM_EXECUTABLE")
	if vim == "" {
		t.Skip("set VIM_EXECUTABLE to the pinned v9.2.1015 executable")
	}
	directory := t.TempDir()
	for name, source := range map[string]string{
		"nested.vim":   "function! Outer()\n  function! Inner()\n    return 'text'\n  endfunction\n  return 42\nendfunction\ncall assert_equal(42, Outer())\n",
		"implicit.vim": "function! NoReturn()\nendfunction\ncall assert_equal(0, NoReturn())\ncall assert_equal(v:t_number, type(NoReturn()))\n",
		"slice.vim":    "vim9script\nvar values = [1, 2, 3]\nvar firstTwo = values[0 : 1]\nassert_equal([1, 2], firstTwo)\nassert_equal(v:t_list, type(firstTwo))\n",
	} {
		if err := os.WriteFile(filepath.Join(directory, name), []byte(source), 0o600); err != nil {
			t.Fatal(err)
		}
	}
	driver := "set nomore\nlet v:errors = []\ntry\n  source nested.vim\n  source implicit.vim\n  source slice.vim\ncatch\n  call add(v:errors, v:exception)\nendtry\ncall assert_equal(902, v:version)\ncall assert_equal(1, has('patch-9.2.1015'))\ncall assert_equal(0, has('patch-9.2.1016'))\nlet report = {'version': v:version, 'patch1015': has('patch-9.2.1015'), 'patch1016': has('patch-9.2.1016'), 'errors': v:errors, 'messages': execute('messages')}\ncall writefile([json_encode(report)], 'result.json')\nif !empty(v:errors)\n  cquit\nendif\nqa!\n"
	if err := os.WriteFile(filepath.Join(directory, "driver.vim"), []byte(driver), 0o600); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	command := exec.CommandContext(ctx, vim, "-Nu", "NONE", "-U", "NONE", "-n", "-es", "-X", "-i", "NONE", "-S", "driver.vim")
	command.Dir = directory
	var stdout, stderr bytes.Buffer
	command.Stdout, command.Stderr = &stdout, &stderr
	err := command.Run()
	report, readErr := os.ReadFile(filepath.Join(directory, "result.json"))
	t.Logf("Vim oracle exit=%v stdout=%q stderr=%q report=%s", err, stdout.String(), stderr.String(), report)
	if err != nil || readErr != nil {
		t.Fatalf("oracle exit=%v report read=%v", err, readErr)
	}
}
